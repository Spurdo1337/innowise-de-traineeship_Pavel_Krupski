// MongoDB aggregation pipelines for reviews_db.google_play_reviews

// 1) Top 5 frequently occurring comments
[
  {
    $group: {
      _id: "$content",
      count: { $sum: 1 }
    }
  },
  {
    $sort: { count: -1 }
  },
  {
    $limit: 5
  },
  {
    $project: {
      _id: 0,
      content: "$_id",
      count: 1
    }
  }
]

// 2) All entries where content is less than 5 characters long
[
  {
    $match: {
      $expr: {
        $lt: [
          { $strLenCP: { $ifNull: ["$content", ""] } },
          5
        ]
      }
    }
  }
]

// 3) Average rating for each day
[
  {
    $group: {
      _id: {
        $dateTrunc: {
          date: "$created_date",
          unit: "day"
        }
      },
      average_rating: { $avg: "$score" }
    }
  },
  {
    $sort: { _id: 1 }
  },
  {
    $project: {
      _id: 0,
      day: "$_id",
      average_rating: 1
    }
  }
]
